fun processImage(image: InputImage) {

    val recognizer = TextRecognition.getClient()

    recognizer.process(image)
        .addOnSuccessListener { visionText ->
            val rawText = visionText.text
            val glucose = extractGlucose(rawText)

            if (glucose != null) {
                onGlucoseDetected(glucose)
            }
        }
}
🔥 Extract số glucose (core AI OCR logic)
fun extractGlucose(text: String): Float? {

    val regex = Regex("""\b(\d{1,2}\.\d)\b""")

    val match = regex.find(text)
    return match?.value?.toFloatOrNull()
}

fun detectSession(hour: Int): Int {
    return when (hour) {
        in 5..7 -> 0 // sáng (fasting)
        in 8..18 -> 1 // trưa
        else -> 2 // chiều/tối
    }
}

🔥 Weighted HbA1c (AI hơn)

👉 Ưu tiên dữ liệu gần đây (giống lab thực tế)

fun weightedAvg(glucose: List<Float>): Float {
    val weights = glucose.mapIndexed { i, _ -> i + 1 }
    val totalWeight = weights.sum()

    return glucose.zip(weights)
        .sumOf { it.first * it.second } / totalWeight
}

🔥 Công thức chuẩn:
fun calculateHbA1c(avg: Float): Float {
    return (weightedAvg + 2.59f) / 1.59f
}
}

fun generateAlert(glucose: Float, hba1c: Float): String {

    return when {
        glucose > 15 -> "🚨 Đường huyết rất cao!"
        hba1c > 9 -> "⚠️ HbA1c nguy hiểm"
        glucose < 3.5 -> "⚠️ Hạ đường huyết"
        else -> "OK"
    }
}
