🧭 2. NAVIGATION (Compose)
sealed class Screen(val route: String) {
    object Dashboard : Screen("dashboard")
    object Detail : Screen("detail/{date}")
    object Input : Screen("input")
    object Stats : Screen("stats")
    object Camera : Screen("camera")
}
@Composable
fun AppNavHost() {
    val navController = rememberNavController()

    NavHost(navController, startDestination = Screen.Dashboard.route) {

        composable(Screen.Dashboard.route) {
            DashboardScreen(navController)
        }

        composable(Screen.Detail.route) {
            DetailScreen()
        }

        composable(Screen.Input.route) {
            InputScreen()
        }

        composable(Screen.Stats.route) {
            StatsScreen()
        }

        composable(Screen.Camera.route) {
            CameraScreen()
        }
    }
}

        }
    }
}
📈 4. CHART (MPAndroidChart)
@Composable
fun GlucoseChart(logs: List<LogEntry>) {

    AndroidView(factory = { context ->
        LineChart(context).apply {

            val entries = logs.mapIndexed { i, log ->
                Entry(i.toFloat(), log.bgBefore?.toFloat() ?: 0f)
            }

            val dataSet = LineDataSet(entries, "Đường huyết").apply {
                lineWidth = 2f
                setDrawCircles(true)
            }

            data = LineData(dataSet)
            invalidate()
        }
    })
}
🧠 5. AI ENGINE (PRO LEVEL – TFLite + fallback rule-based)
class AiEngine @Inject constructor(
    private val context: Context
) {

    fun predict(logs: List<LogEntry>): String {

        if (logs.size < 5) return "Chưa đủ dữ liệu"

        val avg = logs.mapNotNull { it.bgBefore }.average()

        return when {
            avg > 10 -> "⚠️ Đường huyết cao – nguy cơ biến chứng"
            avg < 4 -> "⚠️ Hạ đường huyết"
            else -> "✅ Ổn định"
        }
    }
}

🔔 6. SMART NOTIFICATION (AI)
fun checkAndNotify(context: Context, logs: List<LogEntry>) {

    val latest = logs.lastOrNull() ?: return

    if (latest.bgBefore != null && latest.bgBefore > 12) {

        val notification = NotificationCompat.Builder(context, "health")
            .setContentTitle("⚠️ Cảnh báo đường huyết")
            .setContentText("Giá trị: ${latest.bgBefore}")
            .setSmallIcon(R.drawable.ic_warning)
            .build()

        NotificationManagerCompat.from(context).notify(1, notification)
    }
}

@Composable
fun DetailScreen(
    viewModel: DetailViewModel = hiltViewModel()
) {

    val entries by viewModel.entries.collectAsState()

    LazyColumn {
        items(entries) { entry ->

            SessionCardV2(
                entry = entry,
                onValueChange = {
                    viewModel.update(it) // autosave
                }
            )
        }
    }
}

🧩 10. HILT DI
@Module
@InstallIn(SingletonComponent::class)
object AppModule {

    @Provides
    fun provideDb(app: Application): AppDatabase =
        Room.databaseBuilder(app, AppDatabase::class.java, "db").build()

    @Provides
    fun provideDao(db: AppDatabase) = db.logEntryDao()

    @Provides
    fun provideRepo(dao: LogEntryDao): LogRepository =
        LogRepositoryImpl(dao)
}
